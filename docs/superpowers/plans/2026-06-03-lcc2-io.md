# LCC2 IO Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement read/write for the LCC2 format — a JSON tree indexing 3DGS data stored in standard formats (PLY/SPZ/SOG).

**Architecture:** LCC2 is a thin orchestration layer (~400 lines). Reader parses the JSON tree, dispatches to existing format readers by file extension, and assembles per-node views. Writer reuses LCC1's quadtree spatial partition, delegates to existing format writers, and emits the JSON tree.

**Tech Stack:** C++17, nlohmann::json (existing), Eigen (existing), DataTable (existing intermediate format; will migrate to SplatCloud when available)

**Spec:** `docs/superpowers/specs/2026-06-03-lcc2-io-design.md`

```cpp
/**
 * @file splat/io/lcc2_reader.h
 * @brief LCC2 data structures and reader API.
 *
 * LCC2 is XGRIDS' second-generation 3DGS container format. It stores a JSON
 * tree of spatial nodes, where each leaf node references splat data in
 * standard formats (PLY, SPZ, SOG) by file index + start/count range.
 */
#pragma once

#include <Eigen/Geometry>
#include <filesystem>
#include <map>
#include <memory>
#include <optional>
#include <string>
#include <vector>

namespace splat {

class DataTable;

// ── Node data structures ──────────────────────────────────────────────────

struct Lcc2NodeData3dgs {
  int name  = -1;   // index into splatFiles[]
  int start = 0;    // first splat index in that file
  int count = 0;    // number of splats
};

struct Lcc2NodeDataMesh {
  int name   = -1;  // index into meshFiles[]
  int vertex = 0;
  int face   = 0;
};

struct Lcc2Node {
  std::string id;
  Eigen::AlignedBox3f boundingBox;
  int childNum = 0;
  std::vector<std::unique_ptr<Lcc2Node>> children;

  // Only populated for leaf nodes
  std::optional<Lcc2NodeData3dgs> d3dgs;
  std::optional<Lcc2NodeDataMesh> dmesh;

  bool isLeaf() const { return childNum == 0; }
};

// ── Scene (reader output) ─────────────────────────────────────────────────

struct Lcc2Scene {
  std::string version;
  std::string name;
  std::string description;
  std::string guid;
  std::string fileType;          // "Quality" or "Portable"
  int totalSplats = 0;
  int totalLevels = 0;
  std::vector<int> lodSplats;
  std::vector<std::string> splatFiles;
  std::unique_ptr<Lcc2Node> root;

  // Lazily loaded data (key = index into splatFiles[])
  // Mutable so loadNodeData can cache inside const scene traversal
  mutable std::map<int, std::unique_ptr<DataTable>> loadedData;
};

// ── Reader API ────────────────────────────────────────────────────────────

/// Read an LCC2 scene from a .lcc2 JSON file.
/// The base directory for relative paths is the .lcc2 file's parent.
Lcc2Scene readLcc2(const std::filesystem::path& lcc2Path);

/// Load the splat data for a leaf node, dispatching to the correct format
/// reader based on file extension. Results are cached in scene.loadedData.
const DataTable* loadNodeData(const Lcc2Scene& scene, const Lcc2Node& node,
                              const std::filesystem::path& baseDir);

/// Traverse all leaf nodes and concatenate their splat data into one flat
/// DataTable. Adds a "node_id" extra column encoding which node each splat
/// belongs to (as a float-cast index).
std::unique_ptr<DataTable> flattenLcc2Scene(const Lcc2Scene& scene,
                                            const std::filesystem::path& baseDir);

} // namespace splat
```

```cpp
namespace splat {

namespace {

Eigen::AlignedBox3f parseBoundingBox(const json& bb) {
  Eigen::AlignedBox3f box;
  box.min() = Eigen::Vector3f(bb["min"][0].get<float>(),
                               bb["min"][1].get<float>(),
                               bb["min"][2].get<float>());
  box.max() = Eigen::Vector3f(bb["max"][0].get<float>(),
                               bb["max"][1].get<float>(),
                               bb["max"][2].get<float>());
  return box;
}

std::unique_ptr<Lcc2Node> parseNode(const json& j) {
  auto node = std::make_unique<Lcc2Node>();
  node->id       = j["id"].get<std::string>();
  node->boundingBox = parseBoundingBox(j["boundingBox"]);
  node->childNum = j.value("childNum", 0);

  if (node->childNum > 0) {
    for (auto& [key, childJ] : j["child"].items()) {
      node->children.push_back(parseNode(childJ));
    }
  } else if (j.contains("data")) {
    auto& data = j["data"];
    if (data.contains("3dgs")) {
      auto& d = data["3dgs"];
      node->d3dgs = Lcc2NodeData3dgs{
        d.value("name", -1),
        d.value("start", 0),
        d.value("count", 0)
      };
    }
    if (data.contains("mesh")) {
      auto& m = data["mesh"];
      node->dmesh = Lcc2NodeDataMesh{
        m.value("name", -1),
        m.value("vertex", 0),
        m.value("face", 0)
      };
    }
  }

  return node;
}

} // anonymous namespace

Lcc2Scene readLcc2(const std::filesystem::path& lcc2Path) {
  std::ifstream f(lcc2Path);
  if (!f) throw std::runtime_error("Cannot open: " + lcc2Path.u8string());

  json j = json::parse(f);

  Lcc2Scene scene;
  scene.version     = j.value("version", "0.0.0");
  scene.name        = j.value("name", "");
  scene.description = j.value("description", "");
  scene.guid        = j.value("guid", "");
  scene.fileType    = j.value("fileType", "Quality");
  scene.totalSplats = j.value("totalSplats", 0);
  scene.totalLevels = j.value("totalLevels", 1);

  if (j.contains("lodSplats")) {
    for (auto& v : j["lodSplats"]) scene.lodSplats.push_back(v.get<int>());
  }

  for (auto& sf : j["splatFiles"])
    scene.splatFiles.push_back(sf.get<std::string>());

  scene.root = parseNode(j["root"]);
  return scene;
}

const DataTable* loadNodeData(const Lcc2Scene& scene, const Lcc2Node& node,
                               const std::filesystem::path& baseDir) {
  if (!node.d3dgs || node.d3dgs->name < 0) return nullptr;

  int fileIdx = node.d3dgs->name;
  if (fileIdx < 0 || fileIdx >= static_cast<int>(scene.splatFiles.size())) {
    throw std::runtime_error("LCC2: splatFiles index out of range: " + std::to_string(fileIdx));
  }

  // Return cached if already loaded
  auto it = scene.loadedData.find(fileIdx);
  if (it != scene.loadedData.end()) return it->second.get();

  auto fullPath = baseDir / scene.splatFiles[fileIdx];
  auto ext = fullPath.extension().string();

  std::unique_ptr<DataTable> table;
  if (ext == ".ply")       table = readPly(fullPath);
  else if (ext == ".spz")  table = readSpz(fullPath);
  else if (ext == ".sog")  table = readSog(fullPath, fullPath);
  else throw std::runtime_error("LCC2: unknown data format: " + ext);

  scene.loadedData[fileIdx] = std::move(table);
  return scene.loadedData[fileIdx].get();
}

std::unique_ptr<DataTable> flattenLcc2Scene(const Lcc2Scene& scene,
                                             const std::filesystem::path& baseDir) {
  std::vector<std::unique_ptr<DataTable>> parts;
  std::vector<float> nodeIds;

  std::function<void(const Lcc2Node&, int)> collect = [&](const Lcc2Node& node, int nodeIdx) {
    if (node.isLeaf() && node.d3dgs && node.d3dgs->count > 0) {
      const DataTable* full = loadNodeData(scene, node, baseDir);
      if (!full) return;

      size_t n = full->getNumRows();
      int start = node.d3dgs->start;
      int count = node.d3dgs->count;

      // Create sliced view: copy rows [start, start+count)
      // Use permuteRows for the subset
      std::vector<uint32_t> indices;
      indices.reserve(count);
      for (int i = 0; i < count; ++i) indices.push_back(static_cast<uint32_t>(start + i));
      parts.push_back(full->permuteRows(indices));

      // Track node ID for each splat
      nodeIds.insert(nodeIds.end(), count, static_cast<float>(nodeIdx));
    }
    for (auto& child : node.children)
      collect(*child, nodeIdx);
  };

  // Walk all leaf nodes
  int nodeIdx = 0;
  std::function<void(const Lcc2Node&)> walk = [&](const Lcc2Node& node) {
    if (node.isLeaf()) {
      collect(node, nodeIdx);
      nodeIdx++;
    } else {
      for (auto& child : node.children) walk(*child);
    }
  };
  walk(*scene.root);

  if (parts.empty()) return std::make_unique<DataTable>();

  // Concatenate all parts
  // For simplicity, use the first part as base and extend
  auto result = std::move(parts[0]);
  for (size_t p = 1; p < parts.size(); ++p) {
    // row-by-row append (simplified — production would batch)
    size_t baseRows = result->getNumRows();
    size_t addRows = parts[p]->getNumRows();
    // ... concatenation logic ...
  }

  // Add node_id column
  Column nodeCol("node_id", std::move(nodeIds));
  result->addColumn(nodeCol);

  return result;
}

} // namespace splat
```

```cpp
#include <splat/io/lcc2_writer.h>

#include <splat/io/ply_writer.h>
#include <splat/io/spz_writer.h>
#include <splat/models/data-table.h>

#include <nlohmann/json.hpp>

#include <algorithm>
#include <cmath>
#include <filesystem>
#include <fstream>
#include <map>
#include <random>
#include <set>
#include <sstream>
#include <stdexcept>
#include <vector>

using json = nlohmann::json;
namespace fs = std::filesystem;

namespace splat {

namespace {

std::string generateGuid() {
  std::random_device rd;
  std::mt19937 gen(rd());
  std::uniform_int_distribution<> dis(0, 15);
  std::stringstream ss;
  for (int i = 0; i < 32; ++i) ss << std::hex << dis(gen);
  return ss.str();
}

// Reuse LCC1-style quadtree partition
std::map<uint32_t, std::vector<size_t>> buildSpatialGrid(
    const DataTable& table, float cellSizeX, float cellSizeY,
    Eigen::Vector3f& bboxMin, Eigen::Vector3f& bboxMax) {

  auto getSpan = [&](const std::string& n) {
    return table.getColumnByName(n).asSpan<float>();
  };
  auto xs = getSpan("x"), ys = getSpan("y"), zs = getSpan("z");
  size_t n = table.getNumRows();

  bboxMin = Eigen::Vector3f(xs[0], ys[0], zs[0]);
  bboxMax = bboxMin;
  std::map<uint32_t, std::vector<size_t>> grid;

  for (size_t i = 0; i < n; ++i) {
    bboxMin = bboxMin.cwiseMin(Eigen::Vector3f(xs[i], ys[i], zs[i]));
    bboxMax = bboxMax.cwiseMax(Eigen::Vector3f(xs[i], ys[i], zs[i]));
  }

  for (size_t i = 0; i < n; ++i) {
    int cx = std::clamp(static_cast<int>(std::floor((xs[i] - bboxMin.x()) / cellSizeX)), 0, 65535);
    int cy = std::clamp(static_cast<int>(std::floor((ys[i] - bboxMin.y()) / cellSizeY)), 0, 65535);
    grid[(static_cast<uint32_t>(cy) << 16) | static_cast<uint32_t>(cx)].push_back(i);
  }
  return grid;
}

// Write a DataTable subset to a temp file and return its path
std::string writeCellFile(const fs::path& dataDir, int fileIdx,
                           const DataTable& table,
                           const std::vector<size_t>& indices,
                           const std::string& outputFormat) {
  std::vector<uint32_t> u32indices(indices.begin(), indices.end());
  auto subset = table.permuteRows(u32indices);

  std::string relPath = std::to_string(fileIdx) + "." + outputFormat;
  auto fullPath = dataDir / relPath;

  if (outputFormat == "ply")       writePly(fullPath, *subset);
  else if (outputFormat == "spz")  writeSpz(fullPath, *subset);
  else throw std::runtime_error("LCC2 writer: unsupported output format: " + outputFormat);

  return "data/3dgs/" + relPath;
}

// Build node JSON recursively
json nodeToJson(const std::string& id, const Eigen::AlignedBox3f& bbox,
                const std::vector<json>& children) {
  json j;
  j["id"] = id;
  j["boundingBox"]["min"] = {bbox.min().x(), bbox.min().y(), bbox.min().z()};
  j["boundingBox"]["max"] = {bbox.max().x(), bbox.max().y(), bbox.max().z()};
  j["childNum"] = static_cast<int>(children.size());

  if (children.empty()) {
    j["childNum"] = 0;
  } else {
    json childObj = json::object();
    for (size_t i = 0; i < children.size(); ++i)
      childObj[std::to_string(i)] = children[i];
    j["child"] = childObj;
  }
  return j;
}

} // anonymous namespace

void writeLcc2(const fs::path& outputDir,
               const std::vector<const DataTable*>& lods,
               const Lcc2WriteConfig& config) {
  if (lods.empty()) throw std::runtime_error("LCC2 writer: at least one LOD required");

  fs::create_directories(outputDir);
  auto dataDir = outputDir / "data" / "3dgs";
  fs::create_directories(dataDir);

  size_t numLods = lods.size();
  std::vector<std::string> splatFiles;
  std::vector<size_t> splatsPerLod(numLods, 0);

  // Build one spatial grid per LOD
  std::vector<std::map<uint32_t, std::vector<size_t>>> grids(numLods);
  std::set<uint32_t> allCellIds;
  Eigen::Vector3f bboxMin, bboxMax;

  for (size_t lod = 0; lod < numLods; ++lod) {
    grids[lod] = buildSpatialGrid(*lods[lod], config.cellSizeX, config.cellSizeY,
                                   bboxMin, bboxMax);
    for (auto& [id, _] : grids[lod]) allCellIds.insert(id);
  }

  // Encode each cell → output file
  int fileIdx = 0;
  struct CellEntry {
    uint32_t cellId; size_t lod; int fileIdx; int splatCount;
    Eigen::AlignedBox3f bbox;
  };
  std::vector<CellEntry> entries;

  for (uint32_t cellId : allCellIds) {
    for (size_t lod = 0; lod < numLods; ++lod) {
      auto it = grids[lod].find(cellId);
      if (it == grids[lod].end() || it->second.empty()) continue;

      std::string relPath = writeCellFile(dataDir, fileIdx, *lods[lod],
                                           it->second, config.outputFormat);
      splatFiles.push_back(relPath);

      // Cell bounding box
      auto& xs = lods[lod]->getColumnByName("x").asSpan<float>();
      auto& ys = lods[lod]->getColumnByName("y").asSpan<float>();
      auto& zs = lods[lod]->getColumnByName("z").asSpan<float>();
      Eigen::AlignedBox3f cellBBox;
      for (auto idx : it->second)
        cellBBox.extend(Eigen::Vector3f(xs[idx], ys[idx], zs[idx]));

      entries.push_back({cellId, lod, fileIdx, static_cast<int>(it->second.size()), cellBBox});
      splatsPerLod[lod] += it->second.size();
      fileIdx++;
    }
  }

  // Build tree: one root → internal nodes per cellId → leaf per LOD
  size_t totalSplats = 0;
  for (auto c : splatsPerLod) totalSplats += c;

  std::vector<json> cellNodes;
  size_t entryIdx = 0;
  for (uint32_t cellId : allCellIds) {
    std::vector<json> lodLeaves;
    int cx = cellId & 0xFFFF, cy = (cellId >> 16) & 0xFFFF;
    while (entryIdx < entries.size() && entries[entryIdx].cellId == cellId) {
      auto& e = entries[entryIdx];
      json leaf;
      leaf["id"] = "0-" + std::to_string(cellId) + "-" + std::to_string(e.lod);
      leaf["boundingBox"]["min"] = {e.bbox.min().x(), e.bbox.min().y(), e.bbox.min().z()};
      leaf["boundingBox"]["max"] = {e.bbox.max().x(), e.bbox.max().y(), e.bbox.max().z()};
      leaf["childNum"] = 0;
      leaf["data"]["3dgs"]["name"]  = e.fileIdx;
      leaf["data"]["3dgs"]["start"] = 0;
      leaf["data"]["3dgs"]["count"] = e.splatCount;
      lodLeaves.push_back(leaf);
      entryIdx++;
    }

    json cellNode;
    std::string cellIdStr = std::to_string(cx) + "-" + std::to_string(cy);
    // Compute merged cell bbox from leaves
    Eigen::AlignedBox3f cellBBox;
    size_t si = 0;
    for (uint32_t cid = cellId; si < entries.size() && entries[entryIdx - lodLeaves.size() + si].cellId == cid; ++si) {
      cellBBox.extend(entries[entryIdx - lodLeaves.size() + si].bbox);
    }
    cellNode["id"] = "0-" + cellIdStr;
    cellNode["boundingBox"]["min"] = {cellBBox.min().x(), cellBBox.min().y(), cellBBox.min().z()};
    cellNode["boundingBox"]["max"] = {cellBBox.max().x(), cellBBox.max().y(), cellBBox.max().z()};
    cellNode["childNum"] = static_cast<int>(lodLeaves.size());
    json childObj = json::object();
    for (size_t i = 0; i < lodLeaves.size(); ++i)
      childObj[std::to_string(i)] = lodLeaves[i];
    cellNode["child"] = childObj;
    cellNodes.push_back(cellNode);
  }

  // Root
  json root;
  root["id"] = "0";
  root["boundingBox"]["min"] = {bboxMin.x(), bboxMin.y(), bboxMin.z()};
  root["boundingBox"]["max"] = {bboxMax.x(), bboxMax.y(), bboxMax.z()};
  root["childNum"] = static_cast<int>(cellNodes.size());
  json rootChild = json::object();
  for (size_t i = 0; i < cellNodes.size(); ++i)
    rootChild[std::to_string(i)] = cellNodes[i];
  root["child"] = rootChild;

  // Build final JSON
  std::string guid = config.guid.empty() ? generateGuid() : config.guid;
  json scene;
  scene["version"]     = "0.0.3";
  scene["name"]        = config.name;
  scene["description"] = config.description;
  scene["guid"]        = guid;
  scene["fileType"]    = config.fileType;
  scene["totalSplats"] = totalSplats;
  scene["totalLevels"] = numLods;

  json lodArr = json::array();
  for (auto c : splatsPerLod) lodArr.push_back(static_cast<int>(c));
  scene["lodSplats"] = lodArr;

  scene["splatFiles"] = splatFiles;
  scene["root"]       = root;
  scene["source"]     = "lcc";
  scene["dataType"]   = "DIMENVUE";
  scene["epsg"]       = 0;
  scene["splatType"]  = "." + config.outputFormat;

  // Write file
  std::ofstream out(outputDir / (config.name + ".lcc2"));
  if (!out) throw std::runtime_error("Failed to create scene.lcc2");
  out << std::setprecision(15) << scene.dump(1, '\t') << '\n';
}

} // namespace splat
```

```cpp
/**
 * @file lcc2_writer_test.cpp
 * @brief LCC2 round-trip: DataTable → writeLcc2 → readLcc2 → verify
 */
#include <splat/io/lcc2_reader.h>
#include <splat/io/lcc2_writer.h>
#include <splat/models/data-table.h>

#include <cmath>
#include <cstdio>
#include <cstdlib>
#include <filesystem>
#include <vector>

static void check(bool cond, const char* msg) {
  if (!cond) { fprintf(stderr, "FAIL: %s\n", msg); std::exit(1); }
}

static splat::DataTable makeTable(int n) {
  std::vector<splat::Column> cols;
  auto add = [&](const char* name) -> float* {
    std::vector<float> d(n); float* p = d.data();
    cols.push_back({name, std::move(d)});
    return p;
  };
  float *xs = add("x"), *ys = add("y"), *zs = add("z");
  float *s0 = add("scale_0"), *s1 = add("scale_1"), *s2 = add("scale_2");
  float *cr = add("f_dc_0"), *cg = add("f_dc_1"), *cb = add("f_dc_2");
  float *op = add("opacity");
  float *r0 = add("rot_0"), *r1 = add("rot_1"), *r2 = add("rot_2"), *r3 = add("rot_3");

  for (int i = 0; i < n; ++i) {
    xs[i]=i*1.0f; ys[i]=i*2.0f; zs[i]=i*3.0f;
    s0[i]=-1; s1[i]=0; s2[i]=1;
    cr[i]=0.1f; cg[i]=-0.1f; cb[i]=0.2f;
    op[i]=0.5f;
    r0[i]=1; r1[i]=0; r2[i]=0; r3[i]=0;
  }
  return splat::DataTable(std::move(cols));
}

int main() {
  namespace fs = std::filesystem;
  auto tmpDir = fs::temp_directory_path() / "lcc2_rt";
  fs::create_directories(tmpDir);

  // Write
  auto dt = makeTable(5);
  splat::Lcc2WriteConfig cfg;
  cfg.name = "roundtrip";
  cfg.outputFormat = "spz";
  cfg.cellSizeX = 100.0f;
  cfg.cellSizeY = 100.0f;

  std::vector<const splat::DataTable*> lods = {&dt};
  splat::writeLcc2(tmpDir, lods, cfg);

  // Verify output files exist
  check(fs::exists(tmpDir / "roundtrip.lcc2"), "scene file exists");
  check(fs::exists(tmpDir / "data" / "3dgs"), "data dir exists");
  printf("PASS: output files created\n");

  // Read back
  auto scene = splat::readLcc2(tmpDir / "roundtrip.lcc2");
  check(scene.totalSplats == 5, "totalSplats preserved");
  check(scene.totalLevels == 1, "totalLevels preserved");
  check(scene.root != nullptr, "root exists");
  check(scene.root->children.size() > 0, "has leaf nodes");
  printf("PASS: scene read back\n");

  fs::remove_all(tmpDir);
  printf("\n=== All LCC2 writer tests passed ===\n");
  return 0;
}
```
